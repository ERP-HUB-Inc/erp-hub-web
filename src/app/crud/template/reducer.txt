import reducer from "../reducer";
import Constant from "../../constants/settings/paymentMethod";
import InitialState from "../../../common/reducers/initialState";

export default {
  request: (state = InitialState.request(), action) => {
    const constants = [
      Constant.REQUEST_PAYMENT_METHOD_PENDING,
      Constant.REQUEST_PAYMENT_METHOD_REJECTED,
      Constant.REQUEST_PAYMENT_METHOD_FULFILLED
    ];
    return reducer.request(state, action, constants);
  },
  archive: (state = InitialState.archive(), action) => {
    const constants = [
      Constant.ARCHIVE_PAYMENT_METHOD_PENDING,
      Constant.ARCHIVE_PAYMENT_METHOD_REJECTED,
      Constant.ARCHIVE_PAYMENT_METHOD_FULFILLED
    ];
    return reducer.archive(state, action, constants);
  },
  add: (state = InitialState.add(), action) => {
    const constants = [
      Constant.ADD_PAYMENT_METHOD_PENDING,
      Constant.ADD_PAYMENT_METHOD_REJECTED,
      Constant.ADD_PAYMENT_METHOD_FULFILLED,
      Constant.SHOW_PAYMENT_METHOD_FORM,
      Constant.RESET_PAYMENT_METHOD
    ];
    return reducer.add(state, action, constants);
  },
  update: (state = InitialState.update(), action) => {
    const constants = [
      Constant.UPDATE_PAYMENT_METHOD_PENDING,
      Constant.UPDATE_PAYMENT_METHOD_REJECTED,
      Constant.UPDATE_PAYMENT_METHOD_FULFILLED,
      Constant.SHOW_PAYMENT_METHOD_FORM,
      Constant.RESET_PAYMENT_METHOD
    ];
    return reducer.update(state, action, constants);
  }
};
